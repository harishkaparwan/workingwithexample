@Configuration
@EnableWebSecurity
public class SecurityConfig extends WebSecurityConfigurerAdapter {

    @Override
    protected void configure(HttpSecurity http) throws Exception {
        http
            .csrf()
                // IMPORTANT: allow SAML POSTs without CSRF token
                .ignoringAntMatchers("/saml/**")
            .and()
            .authorizeRequests()
                .antMatchers("/saml/**").permitAll()  // SAML endpoints must be public
                .anyRequest().authenticated()
            .and()
            .httpBasic().disable()
            .formLogin().disable()
            .logout()
                .logoutSuccessUrl("/");

       
    }


}
